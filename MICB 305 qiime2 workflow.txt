# Create a directory for the project and navigate to it
mkdir /work/final
cd /work/final


# Inspect the manifest file for paired-end data 
nano /datasets/project_2/gastric_cancer/gastric_cancer_manifest.tsv


# Import the gastric dataset using a manifest file using the following code:
qiime tools import \
  --type “SampleData[SequencesWithQuality]” \
  --input-format SingleEndFastqManifestPhred33V2 \
  --input-path  /datasets/project_2/gastric_cancer/gastric_cancer_manifest.tsv \
   --output-path ./demux.qza


# Create visualization of demultiplexed samples
qiime demux summarize \
  --i-data demux.qza \
  --o-visualization demux.qzv


# Determine ASVs with DADA2 - denoising + truncation
qiime dada2 denoise-single \
  --i-demultiplexed-seqs demux.qza \
  --p-trim-left 0 \
  --p-trunc-len 252 \
  --o-representative-sequences rep-seqs.qza \
  --o-table table.qza \
  --o-denoising-stats stats.qza
  
# Visualize DADA2 stats
qiime metadata tabulate \
  --m-input-file stats.qza \
  --o-visualization stats.qzv


# Visualize ASVs stats
qiime feature-table summarize \
  --i-table table.qza \
  --o-visualization table.qzv \
  --m-sample-metadata-file /datasets/project_2/gastric_cancer/gastric_cancer_metadata.tsv
 
qiime feature-table tabulate-seqs \
  --i-data rep-seqs.qza \
  --o-visualization rep-seqs.qzv




Train classifier 
# STEP 1: TRIM READS
# silva-138-99-seqs.qza: reference file of all known 16S sequences
# replaced primer sequences with the correct primer sequence from the paper 
# replaced trunc-len with the one defined in the denoising step
qiime feature-classifier extract-reads \
  --i-sequences /datasets/silva_ref_files/silva-138-99-seqs.qza \
  --p-f-primer GTGCCAGCMGCCGCGGTAA \
  --p-r-primer GGACTACHVGGGTWTCTAAT \
  --p-trunc-len 252 \
  --o-reads ref-seqs-trimmed.qza


# STEP 2: MATCH TRIMMED READS WITH KNOWN TAXONOMIES
# Annotate taxonomy with new trimmed classifier (ref-seqs-trimmed.qza)
# /datasets/silva_ref_files/silva-138-99-tax.qza: matches sequences with taxonomy
qiime feature-classifier fit-classifier-naive-bayes \
  --i-reference-reads ref-seqs-trimmed.qza \
  --i-reference-taxonomy /datasets/silva_ref_files/silva-138-99-tax.qza \
  --o-classifier classifier.qza


# STEP 3: APPLY TO YOUR DATASET
# Use the trained classifier to assign taxonomy to reads (rep-seqs.qza)
qiime feature-classifier classify-sklearn \
  --i-classifier classifier.qza \
  --i-reads rep-seqs.qza \
  --o-classification taxonomy.qza


# Turn taxonomy file into readable format 
qiime metadata tabulate \
  --m-input-file taxonomy.qza \
  --o-visualization taxonomy.qzv
  
# Taxonomy barplots
qiime taxa barplot \
  --i-table table.qza \
  --i-taxonomy taxonomy.qza \
  --m-metadata-file /datasets/project_2/gastric_cancer/gastric_cancer_metadata.tsv \
  --o-visualization taxa-bar-plots.qzv


# Taxonomy-based filtering (filtered out mitochondrial/chloroplast sequences and only retained ASVs with at least a phylum level classification)
qiime taxa filter-table \
  --i-table table.qza \
  --i-taxonomy taxonomy.qza \
  --p-exclude mitochondria,chloroplast \
  --o-filtered-table table-no-mitochondria-no-chloroplast.qza


qiime taxa filter-table \
 --i-table table-no-mitochondria-no-chloroplast.qza \
 --i-taxonomy taxonomy.qza \
 --p-include p__ \
 --o-filtered-table table_with_at_least_phylum.qza


# Generate a tree for phylogenetic diversity analyses
qiime phylogeny align-to-tree-mafft-fasttree \
  --i-sequences rep-seqs.qza \
  --o-alignment aligned-rep-seqs.qza \
  --o-masked-alignment masked-aligned-rep-seqs.qza \
  --o-tree unrooted-tree.qza \
  --o-rooted-tree rooted-tree.qza 


# Alpha-rarefaction curve 
qiime diversity alpha-rarefaction \
  --i-table table.qza \
  --i-phylogeny rooted-tree.qza \
  --p-max-depth 8000 \
  --m-metadata-file /datasets/project_2/gastric_cancer/gastric_cancer_metadata.tsv \
  --o-visualization alpha-rarefaction.qzv


# Export 


qiime tools export \
--input-path table_with_at_least_phylum.qza \
--output-path .


qiime tools export \
--input-path taxonomy.qza \
--output-path export


qiime tools export \
--input-path rooted-tree.qza \
--output-path export


biom convert \
-i feature-table.biom \
--to-tsv \
-o exports/feature-table.txt
